# IRIDIS 2.4.0: libiris_sd.so = upstream stable-diffusion.cpp (static, with its own ggml) + sd_jni.cpp.
# Built by scripts/build-sd-native.sh (CI: .github/workflows/native.yml), packaged from prebuilt-native/.
# 2.4.3: IRIS_SDQ=ON builds libiris_sdq.so instead, from the Qualcomm Hexagon branch (scripts/build-sdq-native.sh).
# Only Java_* is exported, so its ggml never clashes with libiris_llama's in the same process.
cmake_minimum_required(VERSION 3.22)
project(iris_sd C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SD_DIR "" CACHE PATH "stable-diffusion.cpp checkout")
option(IRIS_SDQ "Qualcomm Hexagon branch (libiris_sdq.so)" OFF)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${SD_DIR} sd)
if (IRIS_SDQ)
  set(IRIS_SD_TARGET iris_sdq)
else()
  set(IRIS_SD_TARGET iris_sd)
endif()
add_library(${IRIS_SD_TARGET} SHARED sd_jni.cpp)
target_include_directories(${IRIS_SD_TARGET} PRIVATE ${SD_DIR}/include)
if (IRIS_SDQ)
  target_compile_definitions(${IRIS_SD_TARGET} PRIVATE IRIS_SDQ=1)
endif()
target_link_libraries(${IRIS_SD_TARGET} PRIVATE stable-diffusion log android)
target_link_options(${IRIS_SD_TARGET} PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" -Wl,--gc-sections -Wl,-z,max-page-size=16384)
