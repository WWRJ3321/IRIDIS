# 3.0.0 Beta5: libiris_upscale.so = upstream stable-diffusion.cpp (CPU only, its own static ggml) + upscale_jni.cpp.
# Only the ESRGAN upscaler API is used (生图后超分). Packaged in the APK; only Java_* exported.
# Build: scripts/build-upscale-native-arm64host.sh
cmake_minimum_required(VERSION 3.22)
project(iris_upscale C CXX)
set(CMAKE_CXX_STANDARD 17)
# 只留超分用到的代码：分段 + --gc-sections
string(APPEND CMAKE_C_FLAGS " -ffunction-sections -fdata-sections")
string(APPEND CMAKE_CXX_FLAGS " -ffunction-sections -fdata-sections")
set(SD_DIR "" CACHE PATH "stable-diffusion.cpp checkout")
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${SD_DIR} sd)
# 去掉内嵌词表（超分用不到，≈30 MB）
get_target_property(_sd_srcs stable-diffusion SOURCES)
list(FILTER _sd_srcs EXCLUDE REGEX "tokenizers/vocab/vocab\\.cpp$")
set_property(TARGET stable-diffusion PROPERTY SOURCES ${_sd_srcs} ${CMAKE_CURRENT_SOURCE_DIR}/vocab_stub.cpp)
target_include_directories(stable-diffusion PRIVATE ${SD_DIR}/src/tokenizers/vocab)
add_library(iris_upscale SHARED upscale_jni.cpp)
target_include_directories(iris_upscale PRIVATE ${SD_DIR}/include)
if(ANDROID)
  target_link_libraries(iris_upscale PRIVATE stable-diffusion log android)
else()
  target_link_libraries(iris_upscale PRIVATE stable-diffusion)
  add_executable(upscale_hosttest hosttest.cpp)
  target_include_directories(upscale_hosttest PRIVATE ${SD_DIR}/include)
  target_link_libraries(upscale_hosttest PRIVATE stable-diffusion)
endif()
target_link_options(iris_upscale PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/../sdcpp/exports.map" -Wl,--gc-sections -Wl,-z,max-page-size=16384)
